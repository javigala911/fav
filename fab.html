function addDrawnVertex(position) {
drawnCount++;
const vertexId = `v-drawn-${drawnCount}`;
cy.add({
group: 'nodes',
data: { id: vertexId, label: `P${drawnCount}` },
position: { x: position.x, y: position.y },
classes: 'drawn-vertex'
});
const newNode = cy.getElementById(vertexId);
handleNodeClickInDrawMode(newNode);
}
function handleNodeClickInDrawMode(node) {
const nodeId = node.id();
if (drawnNodes.length === 0) {
// Primer nodo de la secuencia
drawnNodes.push(nodeId);
node.addClass('evaluating');
} else {
const lastNodeId = drawnNodes[drawnNodes.length - 1];
// Si selecciona el primer nodo de nuevo y tenemos al menos 2 nodos, cerrar polígono
if (nodeId === drawnNodes[0] && drawnNodes.length >= 3) {
autoClosePolygon();
return;
}
// Conectar con el último nodo dibujado
if (lastNodeId !== nodeId) {
const edgeId = `drawn-edge-${lastNodeId}-${nodeId}`;
// Verificar si la arista ya existe
if (cy.getElementById(edgeId).length === 0) {
cy.add({
group: 'edges',
data: { id: edgeId, source: lastNodeId, target: nodeId },
classes: 'drawn-edge'
});
drawnEdges.push(edgeId);
drawnNodes.push(nodeId);
node.addClass('evaluating');
}
}
}
updateDrawMetrics();
}
function updateDrawMetrics() {
const numVertices = drawnNodes.length;
const numEdges = drawnEdges.length;
document.getElementById('draw-vertices-count').innerText = numVertices;
document.getElementById('draw-edges-count').innerText = numEdges;
const shapeName = SHAPE_NAMES[numEdges] || (numEdges > 2 ? `${numEdges}-Gono` : 'Incompleta');
document.getElementById('draw-shape-detected').innerText = numEdges >= 3 ? shapeName : 'Trazando...';
document.getElementById('btn-close-polygon').disabled = !(numVertices >= 3 && numEdges < numVertices);
document.getElementById('btn-add-to-graph').disabled = !(numEdges >= 3);
}
function autoClosePolygon() {
if (drawnNodes.length >= 3) {
const firstNode = drawnNodes[0];
const lastNode = drawnNodes[drawnNodes.length - 1];
const closingEdgeId = `drawn-edge-${lastNode}-${firstNode}`;
if (cy.getElementById(closingEdgeId).length === 0) {
cy.add({
group: 'edges',
data: { id: closingEdgeId, source: lastNode, target: firstNode },
classes: 'drawn-edge'
});
drawnEdges.push(closingEdgeId);
}
updateDrawMetrics();
}
}
function addDrawnFigureToGraph() {
const numEdges = drawnEdges.length;
if (numEdges < 3) return;
const shapeName = SHAPE_NAMES[numEdges] || `Polígono (${numEdges} aristas)`;
const nodeKey = `fig_${Date.now().toString().slice(-4)}`;
const color = SHAPE_COLORS[numEdges] || '#a855f7';
const shape = SHAPE_STYLES[numEdges] || 'ellipse';
// Guardar nueva figura en la estructura global
figureData[nodeKey] = {
name: shapeName,
edges: numEdges,
color: color,
shape: shape
};
// Calcular posición centroide para reemplazar las marcas de borrador por un único nodo consolidado
let sumX = 0, sumY = 0;
drawnNodes.forEach(id => {
const pos = cy.getElementById(id).position();
sumX += pos.x;
sumY += pos.y;
});
const centerX = sumX / drawnNodes.length;
const centerY = sumY / drawnNodes.length;
// Limpiar borrador temporal
clearDrawing();
// Agregar el nodo consolidado al grafo de Cytoscape
cy.add({
group: 'nodes',
data: {
id: nodeKey,
label: `${shapeName}\n(${numEdges} aristas)`,
edges: numEdges,
figColor: color,
nodeShape: shape
},
position: { x: centerX, y: centerY }
});
// Conectar automáticamente con el último nodo registrado para mantener conectividad en el grafo
const existingKeys = Object.keys(figureData).filter(k => k !== nodeKey);
if (existingKeys.length > 0) {
const randomConnect = existingKeys[Math.floor(Math.random() * existingKeys.length)];
connections.push([nodeKey, randomConnect]);
cy.add({
group: 'edges',
data: {
id: `edge-${nodeKey}-${randomConnect}`,
source: nodeKey,
target: randomConnect
}
});
}
updateStartNodeOptions();
document.getElementById('stat-status').innerText = `¡${shapeName} guardado e integrado al grafo!`;
document.getElementById('stat-status').className = 'font-semibold text-emerald-400';
switchMode('search');
}

drawnEdges.forEach(id => cy.remove(cy.getElementById(id)));
drawnNodes = [];
drawnEdges = [];
drawnCount = 0;
updateDrawMetrics();
}

function getNeighbors(nodeId) {
const neighbors = [];
connections.forEach(([u, v]) => {
if (u === nodeId) neighbors.push({ id: v, edge: `edge-${u}-${v}` });
if (v === nodeId) neighbors.push({ id: u, edge: `edge-${u}-${v}` });
});
return neighbors;
}
function buildExecutionSteps(startNodeId, targetEdges, algo) {
const steps = [];
const visited = new Set();
const parent = {};
const parentEdge = {};
const collection = [startNodeId];
visited.add(startNodeId);
steps.push({
currentNode: null,
visitedNodes: Array.from(visited),
queueState: [...collection],
status: 'Inicializando búsqueda...',
found: false
});
let foundNodeId = null;
while (collection.length > 0) {
const current = algo === 'bfs' ? collection.shift() : collection.pop();
const currentEdges = figureData[current].edges;
steps.push({
currentNode: current,
visitedNodes: Array.from(visited),
queueState: [...collection],
status: `Evaluando ${figureData[current].name} (${currentEdges} aristas)...`,
found: false
});
if (currentEdges === targetEdges) {
foundNodeId = current;
steps.push({
currentNode: current,
visitedNodes: Array.from(visited),
queueState: [...collection],
status: `¡Éxito! Figura encontrada: ${figureData[current].name}`,
found: true
});
break;
}
const neighbors = getNeighbors(current);
for (let neighbor of neighbors) {
if (!visited.has(neighbor.id)) {
visited.add(neighbor.id);
parent[neighbor.id] = current;
parentEdge[neighbor.id] = neighbor.edge;
collection.push(neighbor.id);
steps.push({
currentNode: current,
exploredEdge: neighbor.edge,
visitedNodes: Array.from(visited),
queueState: [...collection],
status: `Agregando vecino: ${figureData[neighbor.id].name}`,
found: false
});
}
}
}
let finalPath = [];
let finalEdges = [];
if (foundNodeId) {
let curr = foundNodeId;
while (curr) {
finalPath.unshift(curr);
if (parentEdge[curr]) finalEdges.unshift(parentEdge[curr]);
curr = parent[curr];
}
}
return { steps, finalPath, finalEdges, foundNodeId };
}
function startSearch() {
resetSearchUI();
const startNode = document.getElementById('select-start').value;
const targetEdges = parseInt(document.getElementById('select-target').value);
const { steps, finalPath, finalEdges } = buildExecutionSteps(startNode, targetEdges, selectedAlgorithm);
executionSteps = steps;
currentStepIndex = 0;
isRunning = true;
isPaused = false;
updateControlsUI();
runAnimationLoop(finalPath, finalEdges);
}
function runAnimationLoop(finalPath, finalEdges) {
if (!isRunning || isPaused) return;
if (currentStepIndex < executionSteps.length) {
renderStep(executionSteps[currentStepIndex]);
currentStepIndex++;
const speed = parseInt(document.getElementById('speed-range').value);
animationTimer = setTimeout(() => runAnimationLoop(finalPath, finalEdges), speed);
} else {
isRunning = false;
if (finalPath && finalPath.length > 0) {
highlightFinalPath(finalPath, finalEdges);
document.getElementById('stat-status').innerText = 'Búsqueda finalizada con éxito';
document.getElementById('stat-status').className = 'font-semibold text-emerald-400';
} else {
document.getElementById('stat-status').innerText = 'Objetivo no alcanzado';
document.getElementById('stat-status').className = 'font-semibold text-rose-400';
}
updateControlsUI();
}
}
function renderStep(step) {
document.getElementById('stat-status').innerText = step.status;
document.getElementById('stat-visited').innerText = `${step.visitedNodes.length} / ${Object.keys(figureData).length}`;
const queueNames = step.queueState.map(id => figureData[id].name.substring(0, 3));
document.getElementById('stat-queue').innerText = queueNames.length > 0 ? queueNames.join(', ') : 'Vacía';
cy.nodes().removeClass('evaluating');
step.visitedNodes.forEach(id => cy.getElementById(id).addClass('visited'));
if (step.currentNode) {
const node = cy.getElementById(step.currentNode);
node.removeClass('visited');
node.addClass('evaluating');
}
if (step.exploredEdge) {
cy.getElementById(step.exploredEdge).addClass('explored');
}
if (step.found) {
cy.getElementById(step.currentNode).removeClass('evaluating').addClass('target-found');
}
}
function highlightFinalPath(finalPath, finalEdges) {
finalPath.forEach(id => {
cy.getElementById(id).removeClass('visited evaluating').addClass('in-path');
});
finalEdges.forEach(edgeId => {
cy.getElementById(edgeId).addClass('path-edge');
});
const pathNames = finalPath.map(id => figureData[id].name);
document.getElementById('stat-path').innerText = pathNames.join(' → ');
}
function togglePause() {
if (!isRunning) return;
isPaused = !isPaused;
const btnPause = document.getElementById('btn-pause');
if (isPaused) {
clearTimeout(animationTimer);
btnPause.innerHTML = '<i class="fa-solid fa-play"></i> Reanudar';
btnPause.className = 'py-2.5 px-4 bg-emerald-600 hover:bg-emerald-500 text-white font-semibold rounded-lg shadow-lg transition flex items-center justify-center gap-2';
document.getElementById('stat-status').innerText = 'Pausado';
} else {
btnPause.innerHTML = '<i class="fa-solid fa-pause"></i> Pausar';
btnPause.className = 'py-2.5 px-4 bg-amber-600 hover:bg-amber-500 text-white font-semibold rounded-lg transition flex items-center justify-center gap-2';
const startNode = document.getElementById('select-start').value;
const targetEdges = parseInt(document.getElementById('select-target').value);
const { finalPath, finalEdges } = buildExecutionSteps(startNode, targetEdges, selectedAlgorithm);
runAnimationLoop(finalPath, finalEdges);
}
updateControlsUI();
}
function stepSearch() {
if (!isRunning) {
const startNode = document.getElementById('select-start').value;
const targetEdges = parseInt(document.getElementById('select-target').value);
const { steps } = buildExecutionSteps(startNode, targetEdges, selectedAlgorithm);
executionSteps = steps;
currentStepIndex = 0;
isRunning = true;
isPaused = true;
updateControlsUI();
}
if (currentStepIndex < executionSteps.length) {
renderStep(executionSteps[currentStepIndex]);
currentStepIndex++;
if (currentStepIndex === executionSteps.length) {
const startNode = document.getElementById('select-start').value;
const targetEdges = parseInt(document.getElementById('select-target').value);
const { finalPath, finalEdges } = buildExecutionSteps(startNode, targetEdges, selectedAlgorithm);
if (finalPath && finalPath.length > 0) {
highlightFinalPath(finalPath, finalEdges);
}
isRunning = false;
updateControlsUI();
}
}
}

isPaused = false;

}

}
